cmake_minimum_required(VERSION 3.21)
project(nr_rk4_stage_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenMP REQUIRED COMPONENTS CXX)
if(NOT DEFINED NR_SHADOW_SOURCE OR NOT EXISTS "${NR_SHADOW_SOURCE}")
  message(FATAL_ERROR "NR_SHADOW_SOURCE must be an identity-bound solver source")
endif()
add_executable(nr-rk4-stage-probe probe.cpp)
target_link_libraries(nr-rk4-stage-probe PRIVATE OpenMP::OpenMP_CXX)
target_compile_definitions(nr-rk4-stage-probe PRIVATE NR_K2_HAS_OPENMP=1
  "NR_SHADOW_SOURCE=\"${NR_SHADOW_SOURCE}\"")
if(MSVC)
  target_compile_options(nr-rk4-stage-probe PRIVATE /W4 /permissive-)
else()
  target_compile_options(nr-rk4-stage-probe PRIVATE -Wall -Wextra -Wpedantic)
endif()
