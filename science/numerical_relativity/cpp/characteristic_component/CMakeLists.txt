cmake_minimum_required(VERSION 3.18)
project(nr_native_characteristic_component LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This qualified component requires Windows x64 MSVC")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ILP64 component requires an x64 executable")
endif()
add_executable(nr-characteristic-component main.cpp)
target_compile_features(nr-characteristic-component PRIVATE cxx_std_17)
set_target_properties(nr-characteristic-component PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(nr-characteristic-component PRIVATE /W4 /permissive- /fp:strict)
