cmake_minimum_required(VERSION 3.18)
project(nr_k2_shadow LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(OpenMP)

add_executable(nr-k2-shadow nr_k2_shadow.cpp)
target_compile_features(nr-k2-shadow PRIVATE cxx_std_17)
if(OpenMP_CXX_FOUND)
  target_link_libraries(nr-k2-shadow PRIVATE OpenMP::OpenMP_CXX)
  target_compile_definitions(nr-k2-shadow PRIVATE NR_K2_HAS_OPENMP=1)
else()
  message(WARNING "OpenMP was not found; building the deterministic serial fallback")
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(nr-fo-topology-test tests/periodic_topology_test.cpp)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(nr-fo-topology-test PRIVATE OpenMP::OpenMP_CXX)
    target_compile_definitions(nr-fo-topology-test PRIVATE NR_K2_HAS_OPENMP=1)
  endif()
  if(MSVC)
    target_compile_options(nr-fo-topology-test PRIVATE /W4 /permissive-)
  else()
    target_compile_options(nr-fo-topology-test PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME fo-periodic-topology COMMAND nr-fo-topology-test)
  add_executable(nr-fo-spatial-order-test tests/spatial_order_test.cpp)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(nr-fo-spatial-order-test PRIVATE OpenMP::OpenMP_CXX)
    target_compile_definitions(nr-fo-spatial-order-test PRIVATE NR_K2_HAS_OPENMP=1)
  endif()
  if(MSVC)
    target_compile_options(nr-fo-spatial-order-test PRIVATE /W4 /permissive-)
  else()
    target_compile_options(nr-fo-spatial-order-test PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME fo-spatial-order COMMAND nr-fo-spatial-order-test)
  add_executable(nr-fo-schwarzschild-domain-test tests/schwarzschild_domain_test.cpp)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(nr-fo-schwarzschild-domain-test PRIVATE OpenMP::OpenMP_CXX)
    target_compile_definitions(nr-fo-schwarzschild-domain-test PRIVATE NR_K2_HAS_OPENMP=1)
  endif()
  if(MSVC)
    target_compile_options(nr-fo-schwarzschild-domain-test PRIVATE /W4 /permissive-)
  else()
    target_compile_options(nr-fo-schwarzschild-domain-test PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME fo-schwarzschild-domain COMMAND nr-fo-schwarzschild-domain-test)
endif()

add_executable(nr-k2-3d-shadow nr_k2_3d_shadow.cpp)
target_compile_features(nr-k2-3d-shadow PRIVATE cxx_std_17)
if(OpenMP_CXX_FOUND)
  target_link_libraries(nr-k2-3d-shadow PRIVATE OpenMP::OpenMP_CXX)
  target_compile_definitions(nr-k2-3d-shadow PRIVATE NR_K2_HAS_OPENMP=1)
else()
  message(WARNING "OpenMP was not found; building the deterministic serial 3-D fallback")
endif()

add_executable(nr-k2-fo-ccz4-shadow nr_k2_fo_ccz4_shadow.cpp)
target_compile_features(nr-k2-fo-ccz4-shadow PRIVATE cxx_std_17)
if(OpenMP_CXX_FOUND)
  target_link_libraries(nr-k2-fo-ccz4-shadow PRIVATE OpenMP::OpenMP_CXX)
  target_compile_definitions(nr-k2-fo-ccz4-shadow PRIVATE NR_K2_HAS_OPENMP=1)
else()
  message(WARNING "OpenMP was not found; building the serial FO-CCZ4 contract fallback")
endif()

if(MSVC)
  target_compile_options(nr-k2-shadow PRIVATE /W4 /permissive-)
  target_compile_options(nr-k2-3d-shadow PRIVATE /W4 /permissive-)
  target_compile_options(nr-k2-fo-ccz4-shadow PRIVATE /W4 /permissive-)
else()
  target_compile_options(nr-k2-shadow PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(nr-k2-3d-shadow PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(nr-k2-fo-ccz4-shadow PRIVATE -Wall -Wextra -Wpedantic)
endif()
