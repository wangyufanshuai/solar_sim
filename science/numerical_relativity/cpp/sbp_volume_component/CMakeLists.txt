function(extract_fixed output first after)
  string(FIND "${frozen_source}" "${first}" start)
  string(FIND "${frozen_source}" "${after}" stop)
  if(start LESS 0 OR stop LESS_EQUAL start)
    message(FATAL_ERROR "Frozen equation extraction markers not found")
  endif()
  math(EXPR count "${stop}-${start}")
  string(SUBSTRING "${frozen_source}" ${start} ${count} block)
  set(${output} "${block}" PARENT_SCOPE)
endfunction()

function(replace_once variable old new)
  string(FIND "${${variable}}" "${old}" found)
  if(found LESS 0)
    message(FATAL_ERROR "Explicit native volume adaptation not found")
  endif()
  string(LENGTH "${old}" width)
  math(EXPR tail_start "${found}+${width}")
  string(SUBSTRING "${${variable}}" ${tail_start} -1 tail)
  string(FIND "${tail}" "${old}" duplicate)
  if(NOT duplicate EQUAL -1)
    message(FATAL_ERROR "Native volume adaptation must be unique")
  endif()
  string(REPLACE "${old}" "${new}" replaced "${${variable}}")
  set(${variable} "${replaced}" PARENT_SCOPE)
endfunction()
