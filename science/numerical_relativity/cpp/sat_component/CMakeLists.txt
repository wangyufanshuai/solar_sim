cmake_minimum_required(VERSION 3.20)
project(nr_native_sat_component LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This qualified component requires Windows x64 MSVC")
endif()
find_package(OpenMP REQUIRED COMPONENTS CXX)
# Reuse authenticated equation generation without modifying the qualified source.
add_subdirectory("../sbp_volume_component" "volume_dependency" EXCLUDE_FROM_ALL)
add_executable(nr-sat-component main.cpp)
target_include_directories(nr-sat-component PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/volume_dependency/generated")
target_compile_features(nr-sat-component PRIVATE cxx_std_17)
set_target_properties(nr-sat-component PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(nr-sat-component PRIVATE /W4 /permissive- /fp:strict)
target_compile_definitions(nr-sat-component PRIVATE NR_K2_HAS_OPENMP=1)
target_link_libraries(nr-sat-component PRIVATE OpenMP::OpenMP_CXX bcrypt)
