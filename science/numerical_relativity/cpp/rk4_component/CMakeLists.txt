cmake_minimum_required(VERSION 3.20)
project(nr_native_rk4_component LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This component requires Windows x64 MSVC")
endif()
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_subdirectory("../sat_component" "sat_dependency" EXCLUDE_FROM_ALL)
add_executable(nr-rk4-component main.cpp)
target_include_directories(nr-rk4-component PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/sat_dependency/volume_dependency/generated")
target_compile_features(nr-rk4-component PRIVATE cxx_std_17)
set_target_properties(nr-rk4-component PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(nr-rk4-component PRIVATE /W4 /permissive- /fp:strict)
target_compile_definitions(nr-rk4-component PRIVATE NR_K2_HAS_OPENMP=1)
target_link_libraries(nr-rk4-component PRIVATE OpenMP::OpenMP_CXX bcrypt)
