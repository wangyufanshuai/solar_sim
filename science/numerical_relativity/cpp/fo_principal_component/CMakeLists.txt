cmake_minimum_required(VERSION 3.18)
project(nr_native_fo_principal_component LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This qualified component requires Windows x64 MSVC")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ILP64 component requires an x64 executable")
endif()
add_executable(nr-fo-principal-component main.cpp)
add_executable(nr-fo-rhs-principal-probe rhs_probe.cpp)
foreach(target nr-fo-principal-component nr-fo-rhs-principal-probe)
  target_compile_features(${target} PRIVATE cxx_std_17)
  set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
  target_compile_options(${target} PRIVATE /W4 /permissive- /fp:strict)
endforeach()
